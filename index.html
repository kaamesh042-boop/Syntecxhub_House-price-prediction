# ============================================================ # SyntecxHub — Project 01: House Price Prediction # Linear Regression · California Housing Dataset # ============================================================ # ── 1. Imports ─────────────────────────────────────────────── import numpy as np import pandas as pd import matplotlib.pyplot as plt import seaborn as sns import joblib from sklearn.datasets import fetch_california_housing from sklearn.model_selection import train_test_split from sklearn.preprocessing import StandardScaler from sklearn.linear_model import LinearRegression from sklearn.metrics import mean_squared_error, r2_score # ── 2. Load Dataset ────────────────────────────────────────── print("=" * 55) print(" SyntecxHub | Project 01 — House Price Prediction") print("=" * 55) data = fetch_california_housing(as_frame=True) df = data.frame.copy() # Use a reproducible 2000-row sample df = df.sample(n=2000, random_state=42).reset_index(drop=True) # Target column is MedHouseVal (in $100k units) → convert to dollars df["Price"] = df["MedHouseVal"] * 100_000 df.drop(columns=["MedHouseVal"], inplace=True) print(f"\n[STEP 1] Dataset loaded") print(f" Shape : {df.shape}") print(f" Features : {list(df.columns[:-1])}") print(f" Target : Price (USD)") # ── 3. Explore & Clean ─────────────────────────────────────── print(f"\n[STEP 2] Exploration & Cleaning") print(f" Missing values : {df.isnull().sum().sum()}") print(f" Avg price : ${df['Price'].mean():,.0f}") print(f" Min price : ${df['Price'].min():,.0f}") print(f" Max price : ${df['Price'].max():,.0f}") print(df.describe().round(2).to_string()) # ── 4. Feature & Target Split ──────────────────────────────── FEATURES = ["MedInc", "HouseAge", "AveRooms", "AveBedrms", "Population", "AveOccup", "Latitude", "Longitude"] TARGET = "Price" X = df[FEATURES] y = df[TARGET] # ── 5. Train / Test Split ──────────────────────────────────── X_train, X_test, y_train, y_test = train_test_split( X, y, test_size=0.20, random_state=42 ) print(f"\n[STEP 3] Train/Test split (80/20)") print(f" Training rows : {X_train.shape[0]}") print(f" Testing rows : {X_test.shape[0]}") # ── 6. Scale Features ──────────────────────────────────────── scaler = StandardScaler() X_train_scaled = scaler.fit_transform(X_train) X_test_scaled = scaler.transform(X_test) print(f"\n[STEP 4] StandardScaler applied") print(f" Means : {dict(zip(FEATURES, scaler.mean_.round(3)))}") print(f" Stds : {dict(zip(FEATURES, scaler.scale_.round(3)))}") # ── 7. Train Model ─────────────────────────────────────────── model = LinearRegression() model.fit(X_train_scaled, y_train) print(f"\n[STEP 5] Linear Regression trained") print(f" Intercept : ${model.intercept_:,.2f}") coef_df = pd.DataFrame({ "Feature": FEATURES, "Coefficient": model.coef_.round(2) }).sort_values("Coefficient", key=abs, ascending=False) print(coef_df.to_string(index=False)) # ── 8. Evaluate ────────────────────────────────────────────── y_pred = model.predict(X_test_scaled) rmse = np.sqrt(mean_squared_error(y_test, y_pred)) r2 = r2_score(y_test, y_pred) mae = np.mean(np.abs(y_test.values - y_pred)) print(f"\n[STEP 6] Model Evaluation") print(f" R² : {r2:.4f}") print(f" RMSE : ${rmse:,.2f}") print(f" MAE : ${mae:,.2f}") # ── 9. Example Predictions ─────────────────────────────────── print(f"\n[STEP 7] Example predictions (first 5 test samples)") results = pd.DataFrame({ "Actual": y_test.values[:5].round(0), "Predicted": y_pred[:5].round(0), "Error": (y_pred[:5] - y_test.values[:5]).round(0) }) print(results.to_string(index=False)) # ── 10. Save Model & Scaler ────────────────────────────────── joblib.dump(model, "linear_regression_model.pkl") joblib.dump(scaler, "standard_scaler.pkl") print(f"\n[STEP 8] Saved model → linear_regression_model.pkl") print(f" Saved scaler → standard_scaler.pkl") # ── 11. Custom Prediction Function ────────────────────────── def predict_price(MedInc=5.5, HouseAge=25, AveRooms=5.4, AveBedrms=1.1, Population=900, AveOccup=2.6, Latitude=37.2, Longitude=-119.1): """ Predict house price from raw feature values. Parameters ---------- MedInc : Median income in block group ($10k units) HouseAge : Median house age (years) AveRooms : Average number of rooms per household AveBedrms : Average number of bedrooms per household Population : Block group population AveOccup : Average household occupancy Latitude : Block group latitude Longitude : Block group longitude Returns ------- float : Estimated house price in USD """ features = np.array([[MedInc, HouseAge, AveRooms, AveBedrms, Population, AveOccup, Latitude, Longitude]]) scaled = scaler.transform(features) price = model.predict(scaled)[0] price = float(np.clip(price, 50_000, 900_000)) return price # ── 12. Plots ──────────────────────────────────────────────── def plot_results(): fig, axes = plt.subplots(1, 3, figsize=(18, 5)) fig.suptitle("SyntecxHub — Project 01: House Price Prediction", fontsize=14, fontweight="bold") # (a) Predicted vs Actual ax = axes[0] ax.scatter(y_test, y_pred, alpha=0.45, color="#6C63FF", s=20) lims = [min(y_test.min(), y_pred.min()), max(y_test.max(), y_pred.max())] ax.plot(lims, lims, "r--", linewidth=1.2, label="Perfect fit") ax.set_xlabel("Actual Price ($)") ax.set_ylabel("Predicted Price ($)") ax.set_title("Predicted vs Actual") ax.legend(fontsize=9) # (b) Residuals distribution ax = axes[1] residuals = y_pred - y_test.values ax.hist(residuals / 1000, bins=30, color="#FF6B9D", edgecolor="white", linewidth=0.4) ax.axvline(0, color="black", linewidth=1.2, linestyle="--") ax.set_xlabel("Residual ($k)") ax.set_ylabel("Count") ax.set_title("Residual Distribution") # (c) Feature coefficients ax = axes[2] colors = ["#00D4AA" if c > 0 else "#FF6B9D" for c in coef_df["Coefficient"]] ax.barh(coef_df["Feature"], coef_df["Coefficient"] / 1000, color=colors) ax.axvline(0, color="black", linewidth=0.8) ax.set_xlabel("Coefficient ($k, scaled)") ax.set_title("Feature Coefficients") plt.tight_layout() plt.savefig("house_price_prediction_plots.png", dpi=150) plt.show() print("\n[PLOTS] Saved → house_price_prediction_plots.png") plot_results() # ── 13. Demo custom prediction ─────────────────────────────── print("\n[DEMO] Custom prediction example:") price = predict_price(MedInc=7.5, HouseAge=15, AveRooms=6.0, AveBedrms=1.0, Population=800, AveOccup=2.4, Latitude=37.8, Longitude=-122.4) print(f" Estimated price: ${price:,.0f}") print("\n✓ Project complete.\n")
